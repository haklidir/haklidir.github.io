"me" -}} {{- $authors = $authors | append (dict "@id" (printf "%s#person" site.Home.Permalink)) -}} {{- else -}} {{- $authors = $authors | append (dict "@type" "Person" "name" .) -}} {{- end -}} {{- end -}} {{- $date := $page.Date.Format "2006-01-02" -}} {{- if eq $p.date_precision "year" }}{{ $date = $page.Date.Format "2006" }}{{ end -}} {{- $data := dict "@context" "https://schema.org" "@type" "ScholarlyArticle" "@id" (printf "%s#article" $page.Permalink) "url" $page.Permalink "name" $page.Title "headline" (truncate 110 "" $page.Title) "author" $authors "datePublished" $date -}} {{- with $pub.name -}} {{- $container := "Periodical" -}} {{- if in (slice "paper-conference" "chapter" "book") $type }}{{ $container = "Book" }}{{ end -}} {{- if eq $type "thesis" }}{{ $container = "" }}{{ end -}} {{- with $container }}{{ $data = merge $data (dict "isPartOf" (dict "@type" . "name" $pub.name)) }}{{ end -}} {{- end -}} {{- with $pub.publisher }}{{ $data = merge $data (dict "publisher" (dict "@type" "Organization" "name" .)) }}{{ end -}} {{- with $pub.pages }}{{ $data = merge $data (dict "pagination" (printf "%v" .)) }}{{ end -}} {{- $same := slice -}} {{- with $ids.doi -}} {{- $data = merge $data (dict "identifier" (dict "@type" "PropertyValue" "propertyID" "DOI" "value" .)) -}} {{- $same = $same | append (printf "https://doi.org/%s" .) -}} {{- end -}} {{- with $ids.arxiv }}{{ $same = $same | append (printf "https://arxiv.org/abs/%s" .) }}{{ end -}} {{- with $ids.acl_id }}{{ $same = $same | append (printf "https://aclanthology.org/%s/" .) }}{{ end -}} {{- with $same }}{{ $data = merge $data (dict "sameAs" .) }}{{ end -}} {{- with $p.abstract }}{{ $data = merge $data (dict "abstract" (. | plainify | htmlUnescape)) }}{{ end -}} {{- with $summary }}{{ $data = merge $data (dict "description" (. | plainify | htmlUnescape)) }}{{ end -}} {{- $kw := $p.keywords | default $p.tags -}} {{- with $kw }}{{ $data = merge $data (dict "keywords" (delimit . ", ")) }}{{ end -}} {{- $data = merge $data (dict "inLanguage" ($p.pub_language | default "en")) -}} {{- with $page.File }}{{ with $page.Resources.GetMatch (printf "%s.pdf" .ContentBaseName) -}} {{- $data = merge $data (dict "encoding" (dict "@type" "MediaObject" "contentUrl" .Permalink "encodingFormat" "application/pdf")) -}} {{- end }}{{ end -}} {{- with $p.concept }}{{ $data = merge $data (dict "subjectOf" (dict "@type" "WebPage" "url" (printf "%sconcepts/%s/" site.Home.Permalink .))) }}{{ end -}} {{- else -}} {{- $featured_image := partial "functions/get_featured_image" $page }} {{ $schema := "Article" }} {{ if eq $page.Type "blog" }} {{ $schema = "BlogPosting" }} {{ end }} {{ $author := partial "functions/get_author_name" $page }} {{ $branding := partialCached "functions/get_branding" $page "branding" }} {{ $publisher := $branding.organization }} {{ $logo_url := partial "functions/get_logo_url" $page -}} {{- end -}}
